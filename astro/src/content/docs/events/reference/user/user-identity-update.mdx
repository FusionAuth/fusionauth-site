import Event from 'src/content/docs/events/reference/_event.astro';
import EventBody from 'src/content/docs/events/reference/_event-body.astro';

export const eventType = 'user.identity.update';

<Event description="This event is generated when a user's phone number is added, changed, or removed."
       eventType={eventType}
       name="User Identity Update"
       scope="tenant"
       transactional="false"
       version="1.59.0"/>

<Aside type="note">
For email address changes, FusionAuth generates the [User Email Update](/docs/events/reference/user/user-email-update) event instead.
</Aside>

<EventBody eventType={eventType}
           jsonFile="user-identity-update.json">

  <APIField slot="leading-fields" name="event.createInstant" type="Long">
    The [instant](/docs/apis/data-types#instants) that the event was generated.
  </APIField>

  <APIField slot="leading-fields" name="event.id" type="UUID">
    The unique Id of the event. You may receive an event more than once based upon your transaction settings. This Id may be used to identify a duplicate event.
  </APIField>

  <APIField slot="trailing-fields" name="event.linkedObjectId" type="UUID">
    The Id of the user impacted by this event.
  </APIField>

  <APIField slot="trailing-fields" name="event.loginIdType" type="String">
    The type of the login Id that changed. The value is always `phoneNumber`.
  </APIField>

  <APIField slot="trailing-fields" name="event.newLoginId" type="String">
    The new phone number. The field is missing when the update removed the phone number.
  </APIField>

  <APIField slot="trailing-fields" name="event.previousLoginId" type="String">
    The previous phone number. The field is missing when the update added a phone number to a user that had none.
  </APIField>

  <APIField slot="trailing-fields" name="event.tenantId" type="UUID">
    The unique tenant identifier. This value may not be returned if not applicable.
  </APIField>

  <APIField slot="trailing-fields" name="event.type" type="String">
    The event type, this value will always be **{eventType}**.
  </APIField>

  <APIField slot="trailing-fields" name="event.user" type="Object">
    The user after the update. See the [Users API](/docs/apis/users) for property definitions and example JSON.
  </APIField>

</EventBody>
