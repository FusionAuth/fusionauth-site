import SearchPreprocessingWarning from "src/content/docs/_shared/_search-preprocessing-warning.mdx";
import RepeatedParameterLimit from 'src/content/docs/_shared/_repeated-parameter-limit.mdx';

{/* parameter_prefix: '' for query params, 'search.' for the request body */}
{/* query_string_request: set for GET query params, unset for the request body */}
{/* database_search_engine_type or elasticsearch_search_engine_type: set one per engine */}

<APIBlock>

  {/* query params sort accurateTotal before expand, the body sorts search.accurateTotal after */}
  <If condition={props.query_string_request && props.elasticsearch_search_engine_type}>
    <APIField name={props.parameter_prefix + 'accurateTotal'} type="Boolean" optional defaults="false" since="1.24.0">
      Set to `true` to return an accurate **total** when more than 10,000 Users match the search.

      By default, the search engine stops counting at 10,000 matches. A search that matches more Users returns a **total** of `10000`. The default count works for pagination and most queries.
    </APIField>
  </If>

  <APIField name="expand" type="Array<String>" optional defaults="[memberships, registrations]" since="1.48.0">
    The User fields to include in the response. Removing `memberships` and `registrations` from the response can improve performance on large searches, especially when you manage Application Roles through Registrations and Group Memberships.

    To return `registrations` without `memberships`, set **expand** to `[registrations]`. To return neither field, set **expand** to `[]`.
  </APIField>

  <If condition={!props.query_string_request && props.elasticsearch_search_engine_type}>
    <APIField name={props.parameter_prefix + 'accurateTotal'} type="Boolean" optional defaults="false" since="1.24.0">
      Set to `true` to return an accurate **total** when more than 10,000 Users match the search.

      By default, the search engine stops counting at 10,000 matches. A search that matches more Users returns a **total** of `10000`. The default count works for pagination and most queries.
    </APIField>
  </If>

  {/* ids is the only field whose type differs between query params and the body */}
  <APIField name={props.parameter_prefix + 'ids'} type={props.query_string_request ? 'UUID' : 'Array<UUID>'} optional>
    <If condition={props.query_string_request}>
      A User Id to retrieve. Repeat the parameter to retrieve multiple Users.
    </If>

    <If condition={!props.query_string_request}>
      A list of User Ids to retrieve.
    </If>

    <If condition={props.database_search_engine_type}>
      When you set **{props.parameter_prefix}ids**, FusionAuth ignores **queryString**. FusionAuth doesn't paginate or sort the results.
    </If>

    <If condition={props.elasticsearch_search_engine_type}>
      When you set **{props.parameter_prefix}ids**, FusionAuth ignores **query** and **queryString**. FusionAuth retrieves the Users by Id without searching the Elasticsearch index, and doesn't paginate or sort the results.
    </If>

    <If condition={props.query_string_request}>
      <RepeatedParameterLimit />
    </If>
  </APIField>

  <APIField name={props.parameter_prefix + 'nextResults'} type="String" optional since="1.48.0" renderif={!!props.elasticsearch_search_engine_type}>
    The token from the **nextResults** field of a previous search response. FusionAuth returns the page that follows the last result of the previous response. Use **nextResults** to page past the Elasticsearch `max_result_window` limit.

    Don't combine **nextResults** with **ids**, **query**, **queryString**, or **sortFields**. Omit **startRow** or set **startRow** to `0`.

    For more information, see [Extended Pagination](/docs/users/search/user-search-with-elasticsearch#extended-pagination).
  </APIField>

  <APIField name={props.parameter_prefix + 'numberOfResults'} type="Integer" optional defaults="25">
    The number of results to return. Used for pagination.

    <If condition={props.elasticsearch_search_engine_type}>
      The sum of **startRow** and **numberOfResults** can't exceed the Elasticsearch `max_result_window` setting, which defaults to 10,000. To page further, use **nextResults**.
    </If>
  </APIField>

  <APIField name={props.parameter_prefix + 'query'} type="String" optional since="1.13.0" renderif={!!props.elasticsearch_search_engine_type}>
    The raw JSON Elasticsearch query to search for Users. When you set **ids**, FusionAuth ignores **query**. When you set **query**, FusionAuth ignores **queryString**.

    To search on `registrations`, use **{props.parameter_prefix}query**. The Elasticsearch mapping defines `registrations` as a [nested data type](https://www.elastic.co/guide/en/elasticsearch/reference/6.3/nested.html), so a **queryString** search on `registrations` returns unexpected results.
  </APIField>

  <If condition={props.database_search_engine_type}>
    <APIField name={props.parameter_prefix + 'queryString'} type="String" optional>
      A search term to match against the following User fields:

      * `firstName`
      * `lastName`
      * `fullName`
      * `email`
      * `phoneNumber` <AvailableSince since="1.59.0" />
      * `username`

      FusionAuth returns every User with a field that contains the search term, ignoring case. Use `*` to match any characters, including none. A value of `*` matches all Users. The database search engine doesn't support regular expressions or field-specific queries.

      When you set **ids**, FusionAuth ignores **queryString**.
    </APIField>
  </If>

  <If condition={props.elasticsearch_search_engine_type}>
    <APIField name={props.parameter_prefix + 'queryString'} type="String" optional>
      The Elasticsearch query string to search for Users. When you set **ids** or **query**, FusionAuth ignores **queryString**.

      The match ignores case. Use the `*` wildcard to search by prefix or suffix. A value of `*` matches all Users. To use an Elasticsearch regular expression, wrap the expression in forward slashes, such as `email:/.*example.com/`.

      <If condition={props.query_string_request}>
        URL-encode whitespace in the query string, such as `%20` for a space.
      </If>

      To search one field, put the field name and a colon before the search term. For example, `email:*fusionauth.io` matches only Users with a `fusionauth.io` email address. For a full list of fields, see [User Search Fields](/docs/users/search/user-fields).

      <SearchPreprocessingWarning/>
    </APIField>
  </If>

  <APIField name={props.parameter_prefix + 'startRow'} type="Integer" optional defaults="0">
    The offset of the first result to return. Used for pagination.
  </APIField>

  <APIField name={props.parameter_prefix + 'sortFields'} type="Array" optional>
    The fields to sort the results by. FusionAuth applies the sort fields in the order you list them.
  </APIField>

  <APIField name={props.parameter_prefix + 'sortFields[x].missing'} type="String" optional defaults="_last" renderif={!!props.elasticsearch_search_engine_type}>
    The value to use when a User has no value for this field. The following special values set the position of those Users:

    * `_first` sorts Users without the field first.
    * `_last` sorts Users without the field last.
  </APIField>

  <APIField name={props.parameter_prefix + 'sortFields[x].name'} type="String" optional>
    The name of the field to sort by. This field is required when you set **sortFields**.

    <If condition={props.database_search_engine_type}>
      The database search engine supports the following fields:

      * `birthDate`
      * `email`
      * `fullName`
      * `id`
      * `insertInstant`
      * `lastLoginInstant`
      * `login`
      * `phoneNumber` <AvailableSince since="1.59.0" />
      * `tenantId`
      * `username`
    </If>

    <If condition={props.elasticsearch_search_engine_type}>
      The search index doesn't support sorting on every User field. The Elasticsearch search engine supports the following fields:

      * `birthDate`
      * `email`
      * `fullName`
      * `id` <AvailableSince since="1.13.0" />
      * `insertInstant`
      * `lastLoginInstant` <AvailableSince since="1.13.0" />
      * `login`
      * `phoneNumber`
      * `registrations.applicationId`
      * `registrations.id`
      * `registrations.insertInstant`
      * `registrations.lastLoginInstant`
      * `registrations.roles`
      * `tenantId`
      * `username`
    </If>
  </APIField>

  <APIField name={props.parameter_prefix + 'sortFields[x].order'} type="String" optional defaults="asc">
    The sort order. The possible values are:

    * `asc`
    * `desc`
  </APIField>
</APIBlock>
