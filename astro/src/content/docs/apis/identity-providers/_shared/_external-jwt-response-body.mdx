<APIBlock>
  <APIField name="identityProvider.applicationConfiguration" type="Map<UUID, Object>">
    The configuration for each Application that has this Identity Provider enabled.
  </APIField>
  <APIField name="identityProvider.applicationConfiguration[applicationId].createRegistration" type="Boolean">
    Whether FusionAuth registers the User for the Application automatically. When `true` and a User with no Registration for the Application logs in through the Identity Provider, FusionAuth creates the Registration.
  </APIField>
  <APIField name="identityProvider.applicationConfiguration[applicationId].enabled" type="Boolean">
    Whether this Identity Provider is enabled for the Application that the `applicationId` key specifies.
  </APIField>

  <IdentityProviderAttributeMappingsResponseParameter />

  <APIField name="identityProvider.claimMap" type="Map<String,String>">
    <ClaimMappings />
  </APIField>
  <APIField name="identityProvider.debug" type="Boolean" since="1.7.3">
    Whether debug is enabled for the Identity Provider. When `true`, FusionAuth creates an Event Log each time FusionAuth reconciles a login through the Identity Provider.
  </APIField>
  <APIField name="identityProvider.defaultKeyId" deprecated>
    The Id of the key that verifies the signature of the JWT. In most cases, the JWT header contains the key identifier. FusionAuth uses the key identifier to look up a key in **`identityProvider.verificationKeyIds`** or in this field.

    <DeprecatedSince since="1.69.0">
      In version `1.69.0` and later, use **`identityProvider.verificationKeyIds`**. For backward compatibility, FusionAuth populates **`identityProvider.defaultKeyId`** with the first entry in **`identityProvider.verificationKeyIds`**.
    </DeprecatedSince>
  </APIField>
  <APIField name="identityProvider.domains" type="Array<String>">
    An array of domains that this Identity Provider manages.
  </APIField>
  <APIField name="identityProvider.enabled" type="Boolean">
    Whether this Identity Provider is enabled. When `false`, FusionAuth disables the IdP globally.
  </APIField>
  <APIField name="identityProvider.headerKeyParameter" type="String">
    The name of the header claim that identifies the public key that verifies the signature. In most cases, the header claim is `kid` or `x5t`.
  </APIField>
  <APIField name="identityProvider.id" type="UUID">
    The unique Id of the Identity Provider.
  </APIField>
  <APIField name="identityProvider.insertInstant" type="Long">
    The [instant](/docs/apis/data-types#instants) that FusionAuth added the Identity Provider to the database.
  </APIField>
  <APIField name="identityProvider.lastUpdateInstant" type="Long">
    The [instant](/docs/apis/data-types#instants) that FusionAuth updated the Identity Provider in the database.
  </APIField>
  <APIField name="identityProvider.lambdaConfiguration.reconcileId" type="UUID" since="1.17.0">
    The unique Id of the lambda that FusionAuth runs during the user reconcile process. The lambda maps custom claims from the external Identity Provider to the FusionAuth user.
  </APIField>

  <IdentityProviderLinkingStrategyResponseParameter idp_linking_strategy={props.idp_linking_strategy}
                                                    idp_since={props.idp_since}
                                                    idp_display_name={props.idp_display_name} />

  <APIField name="identityProvider.name" type="String">
    The name of the Identity Provider.
  </APIField>
  <APIField name="identityProvider.oauth2.authorization_endpoint" type="String">
    The authorization endpoint for this Identity Provider. FusionAuth doesn't use this value. FusionAuth only returns this value in the Lookup Identity Provider API response. During integration, you can use this value to redirect the browser to the OAuth2 authorize endpoint.
  </APIField>
  <APIField name="identityProvider.oauth2.emailClaim" type="String">
    The name of the claim that contains the user's email address. When **`identityProvider.linkingStrategy`** is `LinkByEmail` or `LinkByEmailForExistingUser`, FusionAuth finds or creates the user by this email address. When the JWT contains an email address and **`identityProvider.domains`** has values, the email address must be in one of the domains.
  </APIField>
  <APIField name="identityProvider.oauth2.emailVerifiedClaim" type="String">
    The name of the claim that identifies whether the user's email address is verified.

    When **`identityProvider.linkingStrategy`** is `LinkByEmail` or `LinkByEmailForExistingUser`, and the JWT has this claim with a value of `false`, FusionAuth doesn't create a link. FusionAuth returns an error that says FusionAuth can't create a link with an unverified email address.
  </APIField>
  <APIField name="identityProvider.oauth2.token_endpoint" type="String">
    The token endpoint for this Identity Provider. FusionAuth doesn't use this value. FusionAuth only returns this value in the Lookup Identity Provider API response. During integration, you can use this value to complete the OAuth2 grant workflow.
  </APIField>
  <APIField name="identityProvider.oauth2.uniqueIdClaim" type="String">
    The name of the claim that contains the user's unique user Id.
  </APIField>
  <APIField name="identityProvider.oauth2.usernameClaim" type="String">
    The name of the claim that contains the user's username. FusionAuth only uses this claim when **`identityProvider.linkingStrategy`** is `LinkByUsername` or `LinkByUsernameForExistingUser`.
  </APIField>

  <IdentityProviderSourceResponseParameter />

  <IdentityProviderTenantConfiguration />

  <APIField name="identityProvider.tenantId" type="UUID" since="1.62.0">
    The unique Id of the Tenant. The response includes the value when the Identity Provider is scoped to a Tenant. A Tenant-scoped IdP authenticates users only in the context of the Tenant. You can't change the value after you create the Identity Provider.
  </APIField>
  <APIField name="identityProvider.type" type="String">
    The type of this Identity Provider. The value is always `ExternalJWT`.
  </APIField>
  <APIField name="identityProvider.uniqueIdentityClaim" type="String" deprecated>
    The name of the claim that contains the user's email address. When this field has a value, FusionAuth reads the email address from this claim instead of the claim in **`identityProvider.oauth2.emailClaim`**.
  </APIField>
  <APIField name="identityProvider.verificationKeyIds" type="Array<UUID>" since="1.69.0">
    The Ids of the keys that verify JWT signatures. In most cases, the JWT header contains the key identifier. FusionAuth uses the key identifier to look up a key in this list or in **`defaultKeyId`**.

    Before version `1.69.0`, this field doesn't exist. Use **`identityProvider.defaultKeyId`** instead.
  </APIField>
</APIBlock>
