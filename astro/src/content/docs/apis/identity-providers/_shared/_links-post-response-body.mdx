
<APIBlock>
  <APIField name="identityProviderLink.displayName" type="String">
    An optional human-readable name for this Link, such as an email address, username, or given name. Use this value to identify the User that this Link represents in the remote Identity Provider.

    FusionAuth always overwrites this value during login to reflect the most current information from the IdP. In most cases, this value is an email address or username.
  </APIField>
  <APIField name="identityProviderLink.identityProviderId" type="UUID">
    The unique Id of the Identity Provider.
  </APIField>
  <APIField name="identityProviderLink.identityProviderName" type="String" since="1.47.0">
    The name of the Identity Provider.
  </APIField>
  <APIField name="identityProviderLink.identityProviderType" type="String">
    The type of the Identity Provider.
  </APIField>
  <APIField name="identityProviderLink.identityProviderUserId" type="String">
    The User Id that the Identity Provider provides.
  </APIField>
  <APIField name="identityProviderLink.insertInstant" type="Long">
    The [instant](/docs/apis/data-types#instants) when FusionAuth created the Link.
  </APIField>
  <APIField name="identityProviderLink.lastLoginInstant" type="Long">
    The [instant](/docs/apis/data-types#instants) when the User last logged in with this Identity Provider using this Link.
  </APIField>
  <APIField name="identityProviderLink.tenantId" type="UUID">
    The Id of the Tenant that the User belongs to.
  </APIField>
  <APIField name="identityProviderLink.token" type="String">
    The token that the Identity Provider returns. FusionAuth treats the token as opaque, because the token type varies by IdP. Some IdPs don't return this value. When present, the token is typically a long-lived access or refresh token. For specifics, consult the documentation for the individual IdP.
  </APIField>
  <APIField name="identityProviderLink.userId" type="UUID">
    The unique Id of the FusionAuth User that is linked to the Identity Provider.
  </APIField>
</APIBlock>
