    <ApplicationConfigurationCimdCreateRegistration />
  </APIField>
  <APIField name="identityProvider.applicationConfiguration[applicationId].enabled" type="Boolean" optional defaults="false">

  <APIField name="identityProvider.applicationConfiguration[applicationId].keyId" type="UUID" optional>
    An override of **`identityProvider.keyId`** for the Application.

    An override of **`identityProvider.servicesId`** for the Application.

    An override of **`identityProvider.teamId`** for the Application.
  </APIField>
  <APIField name="identityProvider.bundleId" type="String" optional since="1.43.0">
    The Apple Bundle Id from your Apple Developer account, configured for Sign in with Apple. Native applications use the Bundle Id to Sign in with Apple.

  </APIField>

    Whether the Identity Provider is enabled. When `false`, FusionAuth disables the IdP for every Application.
  </APIField>
  <IdentityProviderAttributeMappingsRequestParameter />
  <APIField name="identityProvider.keyId" type="UUID" required>
    The unique Id of the private key that signs the client secret. Download the key from Apple, then import the key into Key Master.
  </APIField>
  <APIField name="identityProvider.lambdaConfiguration.reconcileId" type="UUID" optional>
    The unique Id of the lambda that maps custom claims from Apple to the FusionAuth User during reconciliation. The lambda type must be `AppleReconcile`.

  </APIField>
  <APIField name="identityProvider.scope" type="String" optional>
    The space-separated scopes that FusionAuth requests from Apple. Set the scopes to `email name`.
  </APIField>
  <APIField name="identityProvider.servicesId" type="String" optional>
    The Apple Services Id from your Apple Developer account, configured for Sign in with Apple. Web applications use the Services Id to Sign in with Apple.

    In `1.43.0` and later, the request must include **`identityProvider.bundleId`**, **`identityProvider.servicesId`**, or both. When you leave out **`identityProvider.bundleId`**, **`identityProvider.servicesId`** is required. Before `1.43.0`, **`identityProvider.servicesId`** is required.
  </APIField>
  <APIField name="identityProvider.teamId" type="String" required>
    The Apple App ID Prefix, also called the Team ID, from your Apple Developer account, configured for Sign in with Apple.
  </APIField>
  <IdentityProviderSourceRequestParameter />
  <IdentityProviderTenantConfiguration />
  <APIField name="identityProvider.tenantId" type="UUID" optional since="1.62.0">
    The unique Id of the Tenant. When you set **`identityProvider.tenantId`**, FusionAuth creates an Identity Provider scoped to the Tenant. Otherwise, FusionAuth creates a global IdP.

    A tenant-scoped IdP authenticates users in the Tenant only. A global IdP works with any Tenant. You can't change **`identityProvider.tenantId`** after FusionAuth creates the IdP.
  </APIField>
  <APIField name="identityProvider.type" type="String" required>
    Set to `Apple`.
  </APIField>
