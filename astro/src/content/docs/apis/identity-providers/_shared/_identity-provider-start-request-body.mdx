    The Id of the Application the user logs in to. The Application must have {props.idp_display_name} login enabled.

  <APIField name="data.requestId" type="String" optional renderif={props.idp_display_name === 'SAML v2'}>

  </APIField>
  <APIField name="data.samlDestination" type="String" optional renderif={props.idp_display_name === 'SAML v2'}>
    The destination to expect in the SAML response, usually the FusionAuth ACS URL. When you set this field and the `Destination` attribute of the SAML response has a different value, FusionAuth applies the **`identityProvider.assertionConfiguration.destination.policy`** of the Identity Provider. When you leave out this field, or the SAML response has no `Destination` attribute, FusionAuth doesn't check the destination.
  </APIField>
  <APIField name="identityProviderId" type="UUID" required>
    The unique Id of the Identity Provider that processes the login request.
  </APIField>
  <APIField name="connectionTestId" type="String" since="1.65.0" optional>

  </APIField>
  <APIField name="loginId" type="String" required renderif={props.idp_display_name === 'HYPR'}>
    The login identifier of the user, which is either the email address or the username.
