<APIBlock>
  <APIField name="access_token" type="String">
    The access token, an encoded JSON Web Token (JWT). FusionAuth writes the access token as an HTTP-only session cookie.
  </APIField>
  <APIField name="refresh_token" type="String">
    The refresh token. FusionAuth writes the refresh token as an HTTP-only persistent cookie. The JWT configuration of the Application, or the global JWT configuration, sets the cookie expiration.
  </APIField>
</APIBlock>
