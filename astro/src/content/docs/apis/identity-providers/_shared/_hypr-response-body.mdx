<APIBlock>
  <APIField type="Map<UUID, Object>" name="identityProvider.applicationConfiguration">
    The configuration for each Application that the Identity Provider is enabled for.
  </APIField>
  <APIField name="identityProvider.applicationConfiguration[applicationId].createRegistration" type="Boolean">
    Whether FusionAuth registers the User for the Application automatically. When `true` and a User with no Registration for the Application logs in through the Identity Provider, FusionAuth creates the Registration.
  </APIField>
  <APIField name="identityProvider.applicationConfiguration[applicationId].enabled" type="Boolean">
    Whether the Identity Provider is enabled for the Application that the **`applicationId`** key names.
  </APIField>
  <APIField name="identityProvider.applicationConfiguration[applicationId].relyingPartyApplicationId" type="String">
    An optional override of **`identityProvider.relyingPartyApplicationId`** for the Application.
  </APIField>
  <APIField name="identityProvider.applicationConfiguration[applicationId].relyingPartyURL" type="String">
    An optional override of **`identityProvider.relyingPartyURL`** for the Application.
  </APIField>

  <IdentityProviderAttributeMappingsResponseParameter />

  <APIField name="identityProvider.debug" type="Boolean">
    Whether debug is enabled for the Identity Provider. When `true`, FusionAuth creates an Event Log each time FusionAuth reconciles a login through the Identity Provider.
  </APIField>
  <APIField name="identityProvider.enabled" type="Boolean">
    Whether the Identity Provider is enabled. When `false`, FusionAuth disables the IdP for every Application.
  </APIField>
  <APIField name="identityProvider.id" type="UUID">
    The Id of the Identity Provider.
  </APIField>
  <APIField name="identityProvider.insertInstant" type="Long">
    The [instant](/docs/apis/data-types#instants) that FusionAuth added the Identity Provider to the FusionAuth database.
  </APIField>
  <APIField name="identityProvider.lastUpdateInstant" type="Long">
    The [instant](/docs/apis/data-types#instants) that FusionAuth updated the Identity Provider in the FusionAuth database.
  </APIField>
  <APIField name="identityProvider.lambdaConfiguration.reconcileId" type="UUID" since="1.17.0">
    The unique Id of the lambda that maps custom claims from the external Identity Provider to the FusionAuth User during reconciliation.
  </APIField>

  <IdentityProviderLinkingStrategyResponseParameter idp_since={props.idp_since} idp_linking_strategy={props.idp_linking_strategy} idp_display_name={props.idp_display_name} />

  <APIField name="identityProvider.name" type="String">
    The display name of the HYPR Identity Provider. FusionAuth uses the name for display only.
  </APIField>
  <APIField name="identityProvider.relyingPartyApplicationId" type="String">
    The Relying Party Application Id that HYPR provides to you.
  </APIField>
  <APIField name="identityProvider.relyingPartyURL" type="String">
    The Relying Party URL that HYPR provides to you.
  </APIField>
  <IdentityProviderSourceResponseParameter />
  <APIField name="identityProvider.tenantId" type="UUID" since="1.62.0">
    The unique Id of the Tenant. The response includes the value when the Identity Provider is scoped to a Tenant. A Tenant-scoped IdP authenticates users only in the context of the Tenant. You can't change the value after you create the Identity Provider.
  </APIField>
  <APIField name="identityProvider.type" type="String">
    The type of the Identity Provider. The value is always `HYPR`.
  </APIField>
