
<APIBlock>
  <APIField name="identityProvider.applicationIds" type="Array<UUID>" since="1.16.0">
    The list of Application Ids that are enabled for this Identity Provider.
  </APIField>
  <APIField name="identityProvider.id" type="UUID">
    The unique Id of the Identity Provider.
  </APIField>
  <APIField name="identityProvider.idpEndpoint" type="String" since="1.16.0">
    The configured SAML v2 endpoint. FusionAuth returns this value only for a SAML v2 Identity Provider. You can use this value to build the SAML request, so you don't need to hard code the endpoint in your application.
  </APIField>
  <APIField name="identityProvider.name" type="String">
    The name of the Identity Provider.
  </APIField>
  <APIField name="identityProvider.oauth2.authorization_endpoint" type="String">
    The OAuth2 Authorize endpoint. You can use this value as the redirect location to begin the authorize workflow, so you don't need to hard code the endpoint in your application. FusionAuth returns this value only for an External JWT or OpenID Connect Identity Provider.
  </APIField>
  <APIField name="identityProvider.oauth2.token_endpoint" type="String">
    The OAuth2 Token endpoint. You can use this value during your integration, so you don't need to hard code the URI in your application. FusionAuth returns this value only for an External JWT or OpenID Connect Identity Provider.
  </APIField>
  <APIField name="identityProvider.tenantId" type="UUID" since="1.62.0">
    The Id of the Tenant that a Tenant-scoped Identity Provider belongs to. The response leaves out the value for a global IdP.
  </APIField>
  <APIField name="identityProvider.type" type="String">
    The type of the Identity Provider.
  </APIField>
</APIBlock>
