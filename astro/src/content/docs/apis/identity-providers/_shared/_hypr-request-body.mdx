<APIBlock>
  <APIField name="identityProvider.applicationConfiguration" type="Map<UUID, Object>" optional>
    The configuration for each Application that the Identity Provider is enabled for.
  </APIField>
  <APIField name="identityProvider.applicationConfiguration[applicationId].createRegistration" type="Boolean" optional defaults="true">
    Whether FusionAuth registers the User for the Application automatically. When `true` and a User with no Registration for the Application logs in through the Identity Provider, FusionAuth creates the Registration.
  </APIField>
  <APIField name="identityProvider.applicationConfiguration[applicationId].enabled" type="Boolean" optional defaults="false">
    Whether the Identity Provider is enabled for the Application that the **`applicationId`** key names.
  </APIField>
  <APIField name="identityProvider.applicationConfiguration[applicationId].relyingPartyApplicationId" type="String" optional>
    An override of **`identityProvider.relyingPartyApplicationId`** for the Application.
  </APIField>
  <APIField name="identityProvider.applicationConfiguration[applicationId].relyingPartyURL" type="String" optional>
    An override of **`identityProvider.relyingPartyURL`** for the Application. The value must be an absolute HTTP or HTTPS URL.
  </APIField>

  <IdentityProviderAttributeMappingsRequestParameter />

  <IdentityProviderDebugRequestParameter idp_since={props.idp_since} />

  <APIField name="identityProvider.enabled" type="Boolean" optional defaults="false">
    Whether the Identity Provider is enabled. When `false`, FusionAuth disables the IdP for every Application.
  </APIField>
  <APIField name="identityProvider.lambdaConfiguration.reconcileId" type="UUID" optional since="1.17.0">
    The unique Id of the lambda that maps custom claims from the external Identity Provider to the FusionAuth User during reconciliation.

    The lambda type must be `HYPRReconcile`.
  </APIField>

  <IdentityProviderLinkingStrategyRequestParameter idp_since={props.idp_since} idp_linking_strategy={props.idp_linking_strategy} idp_display_name={props.idp_display_name}/>

  <APIField name="identityProvider.name" type="String" optional since="1.61.0">
    The display name of the Identity Provider. When the request includes **`identityProviderId`** or **`identityProvider.tenantId`**, **`identityProvider.name`** is required.
  </APIField>
  <APIField name="identityProvider.relyingPartyApplicationId" type="String" required>
    The Relying Party Application Id that HYPR provides to you.
  </APIField>
  <APIField name="identityProvider.relyingPartyURL" type="String" required>
    The Relying Party URL that HYPR provides to you. The value must be an absolute HTTP or HTTPS URL.
  </APIField>
  <IdentityProviderSourceRequestParameter />
  <APIField name="identityProvider.tenantId" type="UUID" optional since="1.62.0">
    The unique Id of the Tenant.

    When you set **`identityProvider.tenantId`**, FusionAuth creates an Identity Provider scoped to the Tenant. Otherwise, FusionAuth creates a global IdP.

    A tenant-scoped IdP authenticates users in the Tenant only. A global IdP works with any Tenant. You can't change **`identityProvider.tenantId`** after FusionAuth creates the IdP.
  </APIField>
  <APIField name="identityProvider.type" type="String" required>
    Set to `HYPR`.
  </APIField>
</APIBlock>
