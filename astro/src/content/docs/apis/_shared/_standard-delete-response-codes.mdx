import TableGrid from 'astro-better-tables/TableGrid.astro';
import Row from 'astro-better-tables/Row.astro';
import Cell from 'astro-better-tables/Cell.astro';

<TableGrid caption="Response Codes" widths="1 6">
  <Row slot="head">
    <Cell header>Code</Cell>
    <Cell header>Description</Cell>
  </Row>
  <Row>
    <Cell>200</Cell>
    <Cell>{props.success_code ? props.success_message : 'The request succeeded.'}</Cell>
  </Row>
  <If condition={props.async_enabled}>
    <Row>
      <Cell>202</Cell>
      <Cell>FusionAuth accepted the request, but the request isn't complete yet. The response is empty.</Cell>
    </Row>
  </If>
  <If condition={!props.no_errors}>
    <Row>
      <Cell>400</Cell>
      <Cell>The request was invalid or malformed. The response contains an [Errors](/docs/apis/errors) JSON object with the specific errors. FusionAuth also returns a `400` when the request needs a paid FusionAuth license and no license is present.</Cell>
    </Row>
  </If>
  <Row>
    <Cell>401</Cell>
    <Cell>The request has no valid Authorization header. The header is missing, or the API key isn't valid. The response is empty. For more information, see [Authentication](/docs/apis/authentication).</Cell>
  </Row>
  <If condition={!props.never_missing}>
    <Row>
      <Cell>404</Cell>
      <Cell>The object you requested doesn't exist. The response is empty.</Cell>
    </Row>
  </If>
  <Row>
    <Cell>500</Cell>
    <Cell>FusionAuth had an internal error and logged a stack trace in the FusionAuth log files. The response is empty.</Cell>
  </Row>
  <If condition={!props.never_search_error}>
    <Row>
      <Cell>503</Cell>
      <Cell>The search index is unavailable or returned an error, so FusionAuth can't complete the request. The response contains a JSON body.</Cell>
    </Row>
  </If>
  <If condition={props.webhook_event}>
    <Row>
      <Cell>504</Cell>
      <Cell>One or more webhook endpoints returned an invalid response or didn't respond. Because of the transaction configuration for the event, FusionAuth can't complete the request. FusionAuth logs a stack trace in the FusionAuth log files.</Cell>
    </Row>
  </If>
</TableGrid>
