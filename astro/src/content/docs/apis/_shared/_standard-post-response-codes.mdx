import TableGrid from 'astro-better-tables/TableGrid.astro';
import Row from 'astro-better-tables/Row.astro';
import Cell from 'astro-better-tables/Cell.astro';

<TableGrid caption="Response Codes" widths="1 6">
  <Row slot="head">
    <Cell header>Code</Cell>
    <Cell header>Description</Cell>
  </Row>
  <Row>
    <Cell>{props.success_code ? props.success_code : 200}</Cell>
    <Cell>{props.success_code ? props.success_message : 'The request succeeded. The response contains a JSON body.'}</Cell>
  </Row>
  <If condition={!props.omit_error_malformed}>
    <Row>
      <Cell>400</Cell>
      <Cell>The request was invalid or malformed. The response contains an [Errors](/docs/apis/errors) JSON object with the specific errors. FusionAuth also returns a `400` when the request needs a paid FusionAuth license and no license is present.</Cell>
    </Row>
  </If>
  <If condition={!props.omit_missing_header}>
    <Row>
      <Cell>401</Cell>
      <Cell>FusionAuth didn't authorize the request. The Authorization header is missing, the API key isn't valid, or the API key doesn't have permission for this endpoint and method. The response is empty. For more information, see [Unauthorized Responses](/docs/apis/authentication#unauthorized-responses).</Cell>
    </Row>
  </If>
  <If condition={!!props.missing_message}>
    <Row>
      <Cell>404</Cell>
      <Cell>{props.missing_message}</Cell>
    </Row>
  </If>
  <If condition={props.conflict_detection}>
    <Row>
      <Cell>409</Cell>
      <Cell>FusionAuth found a conflict while processing the request.</Cell>
    </Row>
  </If>
  <If condition={!props.omit_internal_error}>
    <Row>
      <Cell>500</Cell>
      <Cell>FusionAuth had an internal error and logged a stack trace in the FusionAuth log files. The response usually contains an [Errors](/docs/apis/errors) JSON object with the `[Exception]` general error code. On MySQL, a **`data`** field larger than the database column also returns `500`. For more information, see [Data Field Sizes](/docs/concepts/plans-and-limits/limitations#data-field-sizes).</Cell>
    </Row>
  </If>
  <If condition={!props.never_search_error}>
    <Row>
      <Cell>503</Cell>
      <Cell>The search index is unavailable or returned an error, so FusionAuth can't complete the request. The response contains a JSON body.</Cell>
    </Row>
  </If>
  <If condition={props.webhook_event}>
    <Row>
      <Cell>504</Cell>
      <Cell>One or more webhook endpoints returned an invalid response or didn't respond. Because of the transaction configuration for the event, FusionAuth can't complete the request. FusionAuth logs a stack trace in the FusionAuth log files.</Cell>
    </Row>
  </If>
</TableGrid>
