
<APIBlock>
  <APIField name="messenger.accountSID" type="String" required>
    The Twilio Account SID that FusionAuth uses to connect to the Twilio API. Find the Account SID in your Twilio dashboard.
  </APIField>
  <APIField name="messenger.authToken" type="String" optional>
    The Twilio Auth Token that FusionAuth uses to connect to the Twilio API. Find the Auth Token in your Twilio dashboard.
  </APIField>
  <APIField name="messenger.data" type="Object" optional>
    An object that holds any information you want to store with the Messenger.
  </APIField>
  <APIField name="messenger.debug" type="Boolean" optional defaults="false">
    When this field is `true`, FusionAuth creates a Debug Event Log for each message to help you debug messenger errors.
  </APIField>
  <APIField name="messenger.messageTypes" type="Array<String>" optional defaults='["SMS"]' since="1.65.0">
    The set of message types this messenger can send.

    Supported values include:

    * `SMS`
    * `Voice`
  </APIField>
  <APIField name="messenger.fromPhoneNumber" type="String" optional>
    The Twilio phone number that FusionAuth sends messages from. Find the phone number in your Twilio dashboard.

    This field is required when **`messenger.messageTypes`** includes `Voice`. For SMS, set this field or **`messenger.messagingServiceSid`**.
  </APIField>
  <APIField name="messenger.messagingServiceSid" type="String" optional>
    The Id of a Twilio Messaging Service. Find the Messaging Service SID in your Twilio dashboard.

    When you set this field, FusionAuth sends SMS messages through the Messaging Service instead of from **`messenger.fromPhoneNumber`**. Voice messages always use **`messenger.fromPhoneNumber`**.
  </APIField>
  <APIField name="messenger.name" type="String" required>
    The unique Messenger name.
  </APIField>
  <APIField name="messenger.url" type="String" required>
    The fully qualified URL of the Twilio API endpoint that accepts requests from FusionAuth.
  </APIField>
</APIBlock>
