
<APIBlock>
  <APIField name="messenger.accountSID" type="String">
    The Twilio Account SID that FusionAuth uses to connect to the Twilio API.
  </APIField>
  <APIField name="messenger.authToken" type="String">
    The Twilio Auth Token that FusionAuth uses to connect to the Twilio API.
  </APIField>
  <APIField name="messenger.data" type="Object">
    An object that holds any information you want to store with the Messenger.
  </APIField>
  <APIField name="messenger.debug" type="Boolean">
    When this field is `true`, FusionAuth creates a Debug Event Log for each message to help you debug messenger errors.
  </APIField>
  <APIField name="messenger.fromPhoneNumber" type="String">
    The Twilio phone number that FusionAuth sends messages from.
  </APIField>
  <APIField name="messenger.id" type="UUID">
    The unique Id of the Messenger.
  </APIField>
  <APIField name="messenger.insertInstant" type="Long">
    The [instant](/docs/apis/data-types#instants) that FusionAuth added the Messenger to the FusionAuth database.
  </APIField>
  <APIField name="messenger.lastUpdateInstant" type="Long">
    The [instant](/docs/apis/data-types#instants) that FusionAuth last updated the Messenger in the FusionAuth database.
  </APIField>
  <APIField name="messenger.messageTypes" type="Array<String>" since="1.65.0">
    The set of message types this messenger can send.
  </APIField>
  <APIField name="messenger.messagingServiceSid" type="String">
    The Id of a Twilio Messaging Service. When this field has a value, FusionAuth sends SMS messages through the Messaging Service instead of from **`messenger.fromPhoneNumber`**. Voice messages always use **`messenger.fromPhoneNumber`**.
  </APIField>
  <APIField name="messenger.name" type="String">
    The unique Messenger name.
  </APIField>
  <APIField name="messenger.transport" type="String" since="1.26.0">
    Deprecated. This legacy field remains for backward compatibility. Use **messenger.messageTypes** instead.
  </APIField>
  <APIField name="messenger.type" type="String">
    This field is always `Twilio`.
  </APIField>
  <APIField name="messenger.url" type="String">
    The fully qualified URL of the Twilio API endpoint that accepts requests from FusionAuth.
  </APIField>
</APIBlock>
