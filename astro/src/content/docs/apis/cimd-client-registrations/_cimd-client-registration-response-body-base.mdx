{/*  base_field_name is "registration" for a single registration or "registrations[x]" for search results */}

<APIBlock>
  <APIField name={ props.base_field_name + ".cacheExpirationInstant" } type="Long">
    The [instant](/docs/reference/data-types#instants) when the cached copy of the client's metadata document expires. After this instant, the registration enters the `Grace` state. See [CIMD Registration Caching And Lifecycle](/docs/lifecycle/authenticate-users/oauth/cimd-registration-lifecycle).
  </APIField>
  <APIField name={ props.base_field_name + ".cimdDomainId" } type="UUID">
    The Id of the [CIMD domain](/docs/get-started/core-concepts/types/cimd-domains) the client registered through.
  </APIField>
  <APIField name={ props.base_field_name + ".clientIdUrl" } type="String">
    The client's `client_id` URL.
  </APIField>
  <APIField name={ props.base_field_name + ".clientName" } type="String">
    The `client_name` value from the client's metadata document.
  </APIField>
  <APIField name={ props.base_field_name + ".clientUri" } type="String">
    The `client_uri` value from the client's metadata document.
  </APIField>
  <APIField name={ props.base_field_name + ".data" } type="Object">
    An object that can hold any information about the registration that should be persisted.
  </APIField>
  <APIField name={ props.base_field_name + ".graceExpirationInstant" } type="Long">
    The [instant](/docs/reference/data-types#instants) when the registration's grace period ends. After this instant, the registration enters the `Expired` state, and FusionAuth eventually deletes it. See [The Grace Period](/docs/lifecycle/authenticate-users/oauth/cimd-registration-lifecycle#the-grace-period).
  </APIField>
  <APIField name={ props.base_field_name + ".id" } type="UUID">
    The unique Id of the registration.
  </APIField>
  <APIField name={ props.base_field_name + ".insertInstant" } type="Long">
    The [instant](/docs/reference/data-types#instants) when FusionAuth created the registration.
  </APIField>
  <APIField name={ props.base_field_name + ".jwtPopulateLambdaId" } type="UUID">
    The Id of the JWT populate lambda for this client, as set by the domain's [registration lambda](/docs/extend/code/lambdas/cimd-client-registration). Only present when the registration lambda set a value.
  </APIField>
  <APIField name={ props.base_field_name + ".lastUpdateInstant" } type="Long">
    The [instant](/docs/reference/data-types#instants) when FusionAuth last updated the registration.
  </APIField>
  <APIField name={ props.base_field_name + ".overridesByResource" } type="Object">
    The scope decisions for each `resource` the client has used, keyed by the `resource` value. The domain's [registration lambda](/docs/extend/code/lambdas/cimd-client-registration) makes these decisions, or FusionAuth copies the domain's settings when the domain has no registration lambda.
  </APIField>
  <APIField name={ props.base_field_name + ".overridesByResource[resource].providedScopePolicy" } type="Object">
    The standard OpenID Connect scopes this client may request for the `resource`. See the CIMD domain's **providedScopePolicy** in the [CIMD Domain API](/docs/apis/cimd-domains/create-a-cimd-domain) for the structure.
  </APIField>
  <APIField name={ props.base_field_name + ".overridesByResource[resource].scopes" } type="Array<Object>">
    The custom OAuth scopes this client may request for the `resource`. See the CIMD domain's **scopes** in the [CIMD Domain API](/docs/apis/cimd-domains/create-a-cimd-domain) for the structure.
  </APIField>
  <APIField name={ props.base_field_name + ".redirectURIs" } type="Array<String>">
    The `redirect_uris` values from the client's metadata document.
  </APIField>
  <APIField name={ props.base_field_name + ".state" } type="String">
    The state of the registration, calculated from **cacheExpirationInstant** and **graceExpirationInstant** when FusionAuth builds the response. The possible values are:

    * `Active` - The cached copy of the client's metadata document hasn't expired.
    * `Grace` - The cached copy has expired, but the grace period hasn't ended.
    * `Expired` - The grace period has ended.
  </APIField>
  <APIField name={ props.base_field_name + ".tenantId" } type="UUID">
    The Id of the Tenant the client registered in.
  </APIField>
</APIBlock>
