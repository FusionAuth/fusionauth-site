import APIBlock from 'src/components/api/APIBlock.astro';
import APIField from 'src/components/api/APIField.astro';

<APIBlock>
  <APIField name="JWT duration" required>
    The length of time the issued token (access token and Id token) is valid.  JWT tokens are typically short lived.
  </APIField>
  <APIField name="Access token signing key" optional>
    The key used to sign the access token JWT.
  </APIField>
  <APIField name="Access token verification keys" optional since="1.69.0">
    These are the keys that FusionAuth uses to verify access token signatures. If a verification key is expiring, add its replacement here so either key can be used to verify until the old one expires or is removed. The *Access token signing key* value does not need to be explicitly added here.
  </APIField>
  <APIField name="Id token signing key" optional>
    The key used to sign the Id token JWT.
  </APIField>
  <APIField name="Id token verification keys" optional since="1.69.0">
    These are the keys that FusionAuth uses to verify Id token signatures. If a verification key is expiring, add its replacement here so either key can be used to verify until the old one expires or is removed. The *Id token signing key* value does not need to be explicitly added here.
  </APIField>
</APIBlock>
