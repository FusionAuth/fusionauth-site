import EventBody from 'src/content/docs/extend/events-and-webhooks/events/_event-body.astro';
import ReferenceLink from 'src/content/docs/_shared/_reference-link.mdx';

export const eventType = 'token.exchange.complete';

<Event name="Token Exchange Complete"
       eventType={eventType}
       version="1.70.0"
       plan="enterprise"
       description="This event is generated after FusionAuth records a token exchange attempt, successful or not."
       scope="tenant"
       transactional="false" />

See [OAuth Token Exchange](/docs/lifecycle/authenticate-users/oauth/token-exchange) for how FusionAuth processes an exchange. The event carries the same information as the [Token Exchange audit log](/docs/apis/system/search-the-token-exchange-audit-log) entry for the attempt.

<EventBody eventType={eventType}
           jsonFile="token-exchange-complete.json">

  <APIField slot="leading-fields" name="event.actChain" type="Object">
    The `act` claim of the issued token. Only present when FusionAuth issued a token.
  </APIField>

  <APIField slot="leading-fields" name="event.createInstant" type="Long">
    The <ReferenceLink file="data-types#instants" label="instant" /> that the event was generated.
  </APIField>

  <APIField slot="leading-fields" name="event.exchangerClientId" type="UUID">
    The Id of the exchanger Entity. Despite the name, this value is the Entity's Id, not its OAuth client Id. Absent when client authentication failed for an unknown client.
  </APIField>

  <APIField slot="leading-fields" name="event.id" type="UUID">
    The unique Id of the event. Depending on transaction settings, a webhook can receive the same event more than once. Use this Id to detect duplicates.
  </APIField>

  <APIField slot="trailing-fields" name="event.intent" type="String">
    The value of the `intent` request parameter.
  </APIField>

  <APIField slot="trailing-fields" name="event.issuedToken" type="Object">
    Information about the issued token. Only present when FusionAuth issued a token.
  </APIField>

  <APIField slot="trailing-fields" name="event.issuedToken.delegationSemantics" type="String">
    The kind of exchange. This value is always `DELEGATION`.
  </APIField>

  <APIField slot="trailing-fields" name="event.issuedToken.grantedAudience" type="Array<String>">
    The audiences of the issued token.
  </APIField>

  <APIField slot="trailing-fields" name="event.issuedToken.grantedScope" type="String">
    The space-delimited scopes of the issued token.
  </APIField>

  <APIField slot="trailing-fields" name="event.issuedToken.jti" type="String">
    The `jti` claim of the issued token.
  </APIField>

  <APIField slot="trailing-fields" name="event.outcome" type="String">
    The outcome of the attempt. See the [Token Exchange audit log](/docs/apis/system/search-the-token-exchange-audit-log) for the possible values. When a webhook failed to accept the `token.exchange` event, this value is `WEBHOOK_FAILURE`.
  </APIField>

  <APIField slot="trailing-fields" name="event.policy" type="Object">
    Information about the policy engine's decision. Only present when the policy engine ran.
  </APIField>

  <APIField slot="trailing-fields" name="event.policy.decision" type="String">
    The decision of the policy engine. The possible values are `ALLOW`, `DENY`, and `ERROR`.
  </APIField>

  <APIField slot="trailing-fields" name="event.policy.decisionDurationMs" type="Integer">
    The time in milliseconds the policy engine took to decide.
  </APIField>

  <APIField slot="trailing-fields" name="event.policy.engine" type="String">
    The policy engine that evaluated the exchange. The possible values are `ENTITY` and `LAMBDA`.
  </APIField>

  <APIField slot="trailing-fields" name="event.requestedAudience" type="Array<String>">
    The values of the `audience` and `resource` request parameters.
  </APIField>

  <APIField slot="trailing-fields" name="event.subjectToken" type="Object">
    Information about the subject token. When an attempt fails before FusionAuth validates the subject token, this object only contains **type** and, for some failures, **validationStatus**.
  </APIField>

  <APIField slot="trailing-fields" name="event.subjectToken.iss" type="String">
    The `iss` claim of the subject token.
  </APIField>

  <APIField slot="trailing-fields" name="event.subjectToken.jti" type="String">
    The `jti` claim of the subject token.
  </APIField>

  <APIField slot="trailing-fields" name="event.subjectToken.sub" type="String">
    The `sub` claim of the subject token.
  </APIField>

  <APIField slot="trailing-fields" name="event.subjectToken.type" type="String">
    The value of the `subject_token_type` request parameter.
  </APIField>

  <APIField slot="trailing-fields" name="event.subjectToken.validationStatus" type="String">
    How FusionAuth validated the subject token. The possible values are `AUTHORITATIVE`, `BEST_EFFORT`, and `FAILED`.
  </APIField>

  <APIField slot="trailing-fields" name="event.tenantId" type="UUID">
    The unique Id of the Tenant that processed the exchange.
  </APIField>

  <APIField slot="trailing-fields" name="event.type" type="String">
    The event type. This value is always <code>{eventType}</code>.
  </APIField>

</EventBody>
