export const eventType = 'cimd.registration.success';

<Event name="CIMD Registration Success"
       eventType={eventType}
       version="1.70.0"
       plan="enterprise"
       description="This event is generated when FusionAuth registers a CIMD client or updates its registration. FusionAuth updates a registration when it fetches the client's document again after the cached copy expires, and when the registration lambda evaluates a resource the client hasn't used before. FusionAuth doesn't send this event when it reuses a registration without changing it."
       scope="tenant"
       transactional="false" />

See [OAuth Client Registration](/docs/lifecycle/authenticate-users/oauth/client-registration#client-id-metadata-documents) for how FusionAuth registers CIMD clients, and [CIMD Registration Caching And Lifecycle](/docs/lifecycle/authenticate-users/oauth/cimd-registration-lifecycle) for when it fetches a document again.

<EventBody eventType={eventType}
           jsonFile="cimd-registration-success.json">

  <APIField slot="leading-fields" name="event.cimdClientRegistrationId" type="UUID">
    The unique Id of the CIMD client registration. Use this Id with the [CIMD Client Registration API](/docs/apis/cimd-client-registrations).
  </APIField>

  <APIField slot="leading-fields" name="event.cimdDomainId" type="UUID">
    The unique Id of the [CIMD domain](/docs/get-started/core-concepts/types/cimd-domains) the client registered through.
  </APIField>

  <APIField slot="leading-fields" name="event.clientId" type="String">
    The client's `client_id` URL.
  </APIField>

  <APIField slot="leading-fields" name="event.clientName" type="String">
    The `client_name` value from the client's metadata document.
  </APIField>

  <APIField slot="leading-fields" name="event.createInstant" type="Long">
    The <ReferenceLink file="data-types#instants" label="instant" /> that the event was generated.
  </APIField>

  <APIField slot="leading-fields" name="event.id" type="UUID">
    The unique Id of the event. You may receive an event more than once based upon your transaction settings. This Id may be used to identify a duplicate event.
  </APIField>

  <APIField slot="trailing-fields" name="event.resource" type="String">
    The `resource` the client requested.
  </APIField>

  <APIField slot="trailing-fields" name="event.tenantId" type="UUID">
    The unique Id of the Tenant the client registered in.
  </APIField>

  <APIField slot="trailing-fields" name="event.type" type="String">
    The event type. This value is always <code>{eventType}</code>.
  </APIField>

</EventBody>
